Overview

Takes snapshots of all configured databases and diffs them. The database specific parts are in Adapters. Used by Session, but usable on its own when you don't need tape files.

def diff(before, after)
  before.keys.sort.each_with_object({}) do |db_name, out|
    changed_tables = tables_diff(before.fetch(db_name), after.fetch(db_name))
    out[db_name] = changed_tables unless changed_tables.empty?
  end
end

def excluded?(db_name, name)
  patterns = if name.include?(".")
    @excluded_columns
  else
    @excluded_tables
  end
  Patterns.match?(patterns, "#{db_name}.#{name}")
end

def generated_keys(diff, patterns: [])
  diff.to_h do |db_name, tables|
    db = @dbs.fetch(db_name)
    real_tables = db.tables.map(&:to_s)
    keys = tables.filter_map do |table, changes|
      next unless changes.key?("inserted") && real_tables.include?(table)

      pk = db.schema(table.to_sym).select { |_, info| info[:primary_key] }
      next unless pk.size == 1

      column, info = pk.first
      generated = info[:auto_increment] || Patterns.match?(patterns, "#{db_name}.#{table}.#{column}")
      [table, column.to_s] if generated
    end
    [db_name, keys.to_h]
  end
end

def references(diff, patterns: {})
  diff.to_h do |db_name, tables|
    db = @dbs.fetch(db_name)
    real_tables = db.tables.map(&:to_s)
    references = tables.keys.select { real_tables.include?(_1) }.to_h do |table|
      declared = db.foreign_key_list(table.to_sym).filter_map do |foreign_key|
        [foreign_key[:columns].first.to_s, [db_name, foreign_key[:table].to_s]] if foreign_key[:columns].size == 1
      end
      configured = db.schema(table.to_sym).filter_map do |column, _|
        target = patterns.find { |_, columns| Patterns.match?(columns, "#{db_name}.#{table}.#{column}") }&.first
        [column.to_s, target.split(".", 2)] if target
      end
      [table, declared.to_h.merge(configured.to_h)]
    end
    [db_name, references]
  end
end

def resolve(exclude_tables: @excluded_tables, exclude_columns: @excluded_columns, include_views: @included_views)
  @dbs.to_h do |db_name, db|
    tables = db.tables.map(&:to_s)
    views = view_names(db, db_name, include_views)
    table_names = tables.select { Patterns.match?(exclude_tables, "#{db_name}.#{_1}") }
    column_names = (tables - table_names + views).flat_map do |table|
      excluded_column_names(db_name, table, db.schema(table.to_sym), exclude_columns).map { "#{table}.#{_1}" }
    end
    [db_name, {"excluded" => (table_names + column_names).sort, "views" => views}]
  end
end

def snapshot(previous: nil)
  @dbs.to_h do |name, db|
    tables = checksums(db, name).to_h do |table, sum|
      prev = previous&.dig(name, table)
      unchanged = prev && sum && prev.checksum == sum
      snap = if unchanged
        prev
      else
        dump(db, name, table, sum)
      end
      [table, snap]
    end
    views = view_names(db, name).to_h { [_1, dump(db, name, _1, nil, view: true)] }
    [name, tables.merge(views)]
  end
end
